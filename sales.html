import { db } from "./firebase-config.js"; import { collection, getDocs, query, orderBy } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-firestore.js"; // DOM Elements const salesTableBody = document.getElementById("salesTableBody"); const searchSalesInput = document.getElementById("searchSales"); let allSales = []; /** * Fetch and load sales history from Firestore */ async function loadSalesHistory() { try { const salesQuery = query( collection(db, "sales"), orderBy("timestamp", "desc") ); const querySnapshot = await getDocs(salesQuery); allSales = []; querySnapshot.forEach((doc) => { allSales.push({ id: doc.id, ...doc.data() }); }); renderSalesTable(allSales); } catch (error) { console.error("Error fetching sales history:", error); salesTableBody.innerHTML = ` Failed to load sales entries. Please try again. `; } } /** * Render sales records into the DOM table */ function renderSalesTable(salesData) { if (salesData.length === 0) { salesTableBody.innerHTML = ` No sales records found. `; return; } salesTableBody.innerHTML = salesData .map((sale) => { // Handle Date formatting let formattedDateTime = "N/A"; if (sale.timestamp?.toDate) { formattedDateTime = sale.timestamp.toDate().toLocaleString(); } else if (sale.date) { formattedDateTime = `${sale.date} ${sale.time || ""}`.trim(); } // Items count summary const itemsCount = Array.isArray(sale.items) ? sale.items.length : 0; return ` ${sale.invoiceNo || sale.id} ${sale.customerName || "Walk-in Customer"} ${sale.mobile || "N/A"} ${itemsCount} Item(s) ₹${Number(sale.grandTotal || sale.total || 0).toFixed(2)} ${formattedDateTime} `; }) .join(""); } /** * Search/Filter logic for Invoice, Name, or Mobile Number */ searchSalesInput?.addEventListener("input", (e) => { const searchTerm = e.target.value.toLowerCase().trim(); const filteredSales = allSales.filter((sale) => { const invoice = (sale.invoiceNo || sale.id).toLowerCase(); const customer = (sale.customerName || "").toLowerCase(); const mobile = (sale.mobile || "").toLowerCase(); return ( invoice.includes(searchTerm) || customer.includes(searchTerm) || mobile.includes(searchTerm) ); }); renderSalesTable(filteredSales); }); /** * Attach global scope handlers for Modal and Printing */ window.openModal = function (saleId) { const sale = allSales.find((s) => s.id === saleId); if (!sale) return; const modalHtml = `

💊 MediTrack AI - Invoice

Invoice No: ${sale.invoiceNo || sale.id}

Customer: ${sale.customerName || "Walk-in Customer"}

Mobile: ${sale.mobile || "N/A"}

Purchased Items:

    ${ Array.isArray(sale.items) ? sale.items .map( (item) => `
  • ${item.name} x ${item.quantity || 1} - ₹${( (item.price || 0) * (item.quantity || 1) ).toFixed(2)}
    • ` ) .join("") : "
  • No item details available
    • " }

Total Amount: ₹${Number(sale.grandTotal || sale.total || 0).toFixed(2)}

`; document.body.insertAdjacentHTML("beforeend", modalHtml); }; window.closeModal = function () { const modal = document.getElementById("invoiceModal"); if (modal) modal.remove(); }; window.printInvoice = function () { window.print(); }; // Initialize table on load document.addEventListener("DOMContentLoaded", loadSalesHistory);
